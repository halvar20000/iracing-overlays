# iRacing Overlays — Python dependencies
#
# Install:
#   pip install -r requirements.txt
#
# Notes:
#   - pywin32 is only needed on Windows (iRacing's platform) and is auto-
#     skipped elsewhere via the environment marker. It powers the "Go Live"
#     keystroke feature and the HUD-hide reassert logic in the dashboard.
#   - pillow is only used by iracing_livery.py for TGA→PNG conversion of
#     the on-disk paint cache.
#   - requests is used by iracing_livery.py and iracing_catchup.py for the
#     local iRacing car-render server (optional — both overlays still work
#     without it, with the car images disabled).

flask>=3.0
pyirsdk>=1.3
pillow>=10.0
requests>=2.31
pywin32>=306; sys_platform == "win32"

# Optional — only needed for render_race.py if ffmpeg isn't already
# installed system-wide. Bundles a self-contained ffmpeg binary as a
# Python wheel so Windows users don't have to install ffmpeg manually.
imageio-ffmpeg>=0.4.9

# Driving Mode (DRIVE.bat / drive.py) — on-top overlay windows while driving.
# pywebview hosts the Quali Delta / Track Map HTML overlays as always-on-top
# windows (driving_overlay_window.py); on Windows it uses the Edge WebView2
# runtime (preinstalled on Win10/11). pywin32 (above) powers click-through.
pywebview>=5.0; sys_platform == "win32"

# Race Control (iCASControl) — racecontrol/ subfolder, port 8080. Its FastAPI
# steward backend. Only needed if you run the "racectrl" overlay. (racecontrol
# has its own racecontrol/requirements.txt with the same pins.)
fastapi>=0.110
uvicorn>=0.29
websockets>=12.0
